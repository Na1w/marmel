+    agent: &dyn Specialist,
+    expected: Agent,
+    must_contain: &[&str],
+    must_not_contain: &[&str],

+) {

+        assert!(

+            expected

+            expected
+        );
+    }
+    assert_eq!(
+        agent.may_recurse(),
+        may_recurse,
+        "{:?} may_recurse mismatch",
+        expected
+    );
